# `Lib/shutil.py`

<CpyLink path="Lib/shutil.py" />

`shutil` ("shell utilities") is a pure-Python module that provides
operations that the standard `os` module deliberately omits: multi-file
copy with metadata preservation, recursive directory copy and deletion,
cross-filesystem move, archive creation and unpacking, and disk-space
queries.

The module is organized into three broad areas. The first covers file
copying (`copyfileobj`, `copyfile`, `copy`, `copy2`, `copymode`,
`copystat`, `copytree`). The second covers removal and move (`rmtree`,
`move`). The third covers archives (`make_archive`, `unpack_archive`,
`get_archive_formats`, `register_archive_format`) and utilities
(`disk_usage`, `chown`, `which`, `get_terminal_size`).

Platform-specific fast paths are selected at import time. On Linux,
`copyfile` tries `os.sendfile` first and falls back to a `mmap`-based
copy; on macOS it uses `fcopyfile` via `os.copy_file_range`. The
`_USE_CP_SENDFILE` flag records whether the sendfile path is available.
`rmtree` uses `os.open` with `O_PATH` on Linux to avoid symlink races
when `dir_fd` is supported.

def copyfile(src, dst, *, follow_symlinks=True):
    """Copy data from src to dst in the most efficient way possible."""
    sys.audit("shutil.copyfile", src, dst)
    if _samefile(src, dst):
        raise SameFileError("{!r} and {!r} are the same file".format(src, dst))
    file_size = 0
    for fn in [src, dst]:
        try:
            st = _stat(fn)
        except OSError:
            pass
        else:
            if stat.S_ISFIFO(st.st_mode):
                fn_type = "fifo"
            elif stat.S_ISSOCK(st.st_mode):
                fn_type = "socket"
            ...
    with open(src, 'rb') as fsrc:
        try:
            with open(dst, 'wb') as fdst:
                _copyfileobj_readinto(fsrc, fdst, file_size,
                                      _COPY_BUFSIZE)
        except IsADirectoryError as e:
            ...
```

`copyfile` raises `SameFileError` before opening either file to avoid
truncating `dst` when `src` and `dst` are the same path. Special file
types (FIFO, socket, device) are rejected with a descriptive error. The
fast path selection happens inside `_copyfileobj_readinto`: on Linux it
attempts `os.sendfile` in a loop (handling `BlockingIOError` and short
writes), falls back to `os.copy_file_range`, then to a `readinto`-based
loop using a pre-allocated `bytearray` buffer to minimise allocations.

def rmtree(path, ignore_errors=False, onerror=None, *,
           onexc=None, dir_fd=None):
    sys.audit("shutil.rmtree", path)
    if ignore_errors:
        def onexc(*args):
            pass
    elif onerror is None and onexc is None:
        onexc = _rmtree_default_onexc
    ...
    if _use_fd_functions:
        # While the unsafe rmtree works fine on bytes, the fd based does not.
        if isinstance(path, bytes):
            path = os.fsdecode(path)
        # Note: To guard against symlink races, we use the fd-based APIs
        try:
            fd = os.open(path, os.O_RDONLY, dir_fd=dir_fd)
        except Exception as err:
            onexc(os.lstat, path, err)
            return
        return _rmtree_safe_fd(fd, path, onexc)
    else:
        return _rmtree_unsafe(path, onexc)
```

The key insight is that `os.unlink(name, dir_fd=parent_fd)` and
`os.rmdir(name, dir_fd=parent_fd)` operate on the file that is actually
inside `parent_fd`, not on whatever `path` resolves to at the time of the
call. This makes the deletion atomic with respect to concurrent renames or
symlink swaps. The `_use_fd_functions` flag is `True` on platforms where
`os.open`, `os.unlink`, and `os.rmdir` all accept `dir_fd`. On Windows
and older POSIX the unsafe path is used. The `onerror` parameter is
deprecated in 3.12 in favour of `onexc`.

def copytree(src, dst, symlinks=False, ignore=None, copy_function=copy2,
             ignore_dangling_symlinks=False, dirs_exist_ok=False):
    """Recursively copy a directory tree and return the destination directory."""
    sys.audit("shutil.copytree", src, dst)

```

`copytree` snapshots the directory listing with `os.scandir` before
creating `dst`, so the ignore callback sees the full list up front.
`dirs_exist_ok=True` (added in 3.8) passes `exist_ok=True` to
`makedirs`, enabling merging into an existing tree. Errors during
individual copies are collected into a list and raised together as a
single `Error` (a subclass of `OSError`) at the end so that one bad file
does not abort the entire tree copy.

def which(name, mode=os.F_OK | os.X_OK, path=None):
    """Given a name, mode, and path string, return the path which
    conforms to the given mode on the PATH, or None if there is no such
    file."""
    ...
    if os.path.dirname(name):
        # name is a path already
        if _access_check(name, mode):
            return name
        return None
    path_list = (path or os.environ.get('PATH', None) or os.defpath).split(os.pathsep)
    ...
    if sys.platform == 'win32':
        pathext_source = os.environ.get('PATHEXT', '')
        pathext = [ext.lower() for ext in pathext_source.split(os.pathsep)]
        if not any(name.lower().endswith(ext) for ext in pathext):
            files = [name + ext for ext in pathext]
        else:
            files = [name]
    else:
        files = [name]
    for directory in path_list:
        for thefile in files:
            full_path = os.path.join(directory, thefile)
            if _access_check(full_path, mode):
                return full_path
    return None
```

`which` first checks whether `name` contains a path separator; if so it
tests it directly and returns `None` on failure. Otherwise it splits
`PATH` (defaulting to `os.defpath`) and probes each directory. On
Windows, `PATHEXT` is consulted to generate candidate filenames by
appending each extension, matching the behaviour of `cmd.exe`. The
`mode` parameter defaults to `F_OK | X_OK` but callers can pass `F_OK`
alone to locate non-executable files.
