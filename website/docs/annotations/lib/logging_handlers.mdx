# `Lib/logging/handlers.py`

<CpyLink path="Lib/logging/handlers.py" />

`Lib/logging/handlers.py` is the optional extension layer for the
logging framework. It is not imported by `logging/__init__.py`; callers
must `import logging.handlers` explicitly. The file provides handlers
for every common transport: rotating files, syslog, SMTP, TCP/UDP
sockets, in-memory buffers, and asynchronous queues.

All handler classes inherit from `logging.Handler` (or from
`logging.StreamHandler` for stream-based derivatives). The base class
provides `handle(record)`, which acquires the handler lock, calls
`emit(record)`, and releases; subclasses only need to implement
`emit`. The `close()` method in the base unregisters the handler from
the internal `_handlerList` weak-reference registry.

File-based handlers (`RotatingFileHandler`, `TimedRotatingFileHandler`)
inherit from `BaseRotatingHandler`, which adds `shouldRollover()` and
`doRollover()` abstractions and calls them from an overridden `emit`.

```python
class RotatingFileHandler(BaseRotatingHandler):
    def __init__(self, filename, mode='a', maxBytes=0, backupCount=0,
                 encoding=None, delay=False, errors=None):
        if maxBytes > 0:
            mode = 'a'
        BaseRotatingHandler.__init__(self, filename, mode, encoding,
                                     delay, errors)
        self.maxBytes = maxBytes
        self.backupCount = backupCount

    def doRollover(self):
        if self.stream:
            self.stream.close()
            self.stream = None
        if self.backupCount > 0:
            for i in range(self.backupCount - 1, 0, -1):
                sfn = self.rotation_filename("%s.%d" % (self.baseFilename, i))
                dfn = self.rotation_filename("%s.%d" % (self.baseFilename, i + 1))
                if os.path.exists(sfn):
                    if os.path.exists(dfn):
                        os.remove(dfn)
                    os.rename(sfn, dfn)
            dfn = self.rotation_filename(self.baseFilename + ".1")
            if os.path.exists(dfn):
                os.remove(dfn)
            self.rotate(self.baseFilename, dfn)
        if not self.delay:
            self.stream = self._open()

    def shouldRollover(self, record):
        if self.stream is None:
            self.stream = self._open()
        if self.maxBytes > 0:
            msg = "%s\n" % self.format(record)
            self.stream.seek(0, 2)
            if self.stream.tell() + len(msg) >= self.maxBytes:
                return True
        return False
```

The rename loop walks backwards from `backupCount-1` down to 1 so that
`.1` is always renamed to `.2` before the base file is renamed to `.1`.
`rotation_filename` is a hook that lets callers apply a namer callable
(e.g. to add a timestamp suffix). `rotate` is a second hook that lets
callers apply a rotator callable (e.g. for compression).

`shouldRollover` opens the stream if it is `None` (lazy open case),
then seeks to the end and checks whether adding the formatted record
would exceed `maxBytes`. Seeking on every record is cheap because the
OS kernel returns the position from the inode without a syscall on most
platforms.

```python
class SocketHandler(logging.Handler):
    def makePickle(self, record):
        ei = record.exc_info
        if ei:
            dummy = self.format(record)  # force exc_text
        d = dict(record.__dict__)
        d['msg'] = record.getMessage()
        d['args'] = None
        d.pop('exc_info', None)
        s = pickle.dumps(d, 1)
        slen = struct.pack(">L", len(s))
        return slen + s

    def send(self, s):
        if self.sock is None:
            self.sock = self.makeSocket()
        self.sock.sendall(s)

    def emit(self, record):
        try:
            s = self.makePickle(record)
            self.send(s)
        except Exception:
            self.handleError(record)
```

The wire format is a 4-byte big-endian length followed by
`pickle.dumps(record.__dict__, protocol=1)`. The dict form is used
rather than pickling the `LogRecord` directly because many `LogRecord`
attributes (like `exc_info`) are not picklable. The `makeLogRecord`
function on the receiving end reconstructs the `LogRecord` from the
dict.

`DatagramHandler` overrides `send` to use `socket.sendto` instead of
`sendall`, with no reconnection logic. `SocketHandler` retries the
connection after a configurable `retryStart` / `retryMax` backoff.

```python
class QueueHandler(logging.Handler):
    def __init__(self, queue):
        logging.Handler.__init__(self)
        self.queue = queue

    def enqueue(self, record):
        self.queue.put_nowait(record)

    def prepare(self, record):
        msg = self.format(record)
        record = copy.copy(record)
        record.message = msg
        record.msg = msg
        record.args = None
        record.exc_info = None
        record.exc_text = None
        return record

    def emit(self, record):
        try:
            self.enqueue(self.prepare(record))
        except Exception:
            self.handleError(record)


class QueueListener:
    def __init__(self, queue, *handlers, respect_handler_level=False):
        self.queue = queue
        self.handlers = handlers
        self._thread = None
        self.respect_handler_level = respect_handler_level

    def start(self):
        self._thread = t = threading.Thread(target=self._monitor,
                                            daemon=True)
        t.start()

    def _monitor(self):
        while True:
            record = self.dequeue(True)
            if record is self._sentinel:
                break
            self.handle(record)
```

`QueueHandler.prepare` formats the record and clears `exc_info` before
enqueueing, because `exc_info` contains live traceback objects that
cannot safely cross thread or process boundaries (and may keep large
object graphs alive). The formatted string is stored in `record.msg`
so the downstream handlers can emit it without reformatting.

`QueueListener._monitor` loops on `queue.get()` until it receives a
sentinel object (the class itself, `_sentinel = object()`). `stop()`
enqueues the sentinel and calls `thread.join()`. `respect_handler_level`
is a 3.12 addition that skips calling a handler if `record.levelno` is
below `handler.level`, matching the filtering behavior of direct handler
attachment.
